    ['TC-02', 3,  true ],   // ขอบล่าง
    ['TC-03', 15, true ],   // ตัวแทน
    ['TC-04', 30, true ],   // ขอบบน
    ['TC-05', 31, false],   // เลยขอบบน
  ])('%s ขอล่วงหน้า %i วัน ควรผ่าน=%s', (id, days, expected) => {
    const input = { title: 'สัมมนา', daysAhead: days, items: [room] }; // Arrange
    const result = validateRequest(input, resources);                // Act
    expect(result.ok).toBe(expected);                                // Assert
  });
});
