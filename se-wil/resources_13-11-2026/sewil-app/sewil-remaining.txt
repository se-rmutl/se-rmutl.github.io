/**
 * Letter.gs — หนังสือแนะนำรายวิชาและแบบตอบรับ SE-WIL00 ฉบับมีข้อมูล (v1.7)
 *
 * ใช้หน้าแบบฟอร์มเดิมในพอร์ทัล (forms/intro-letter.html) ระบบส่งข้อมูลไปกับลิงก์ส่วนหลัง # ซึ่งเบราว์เซอร์
 * ไม่ส่งไปที่เซิร์ฟเวอร์ของพอร์ทัล หน้าเว็บเติมข้อมูลลงช่องแล้วพิมพ์ ข้อมูลเป็น JSON ที่ gzip แล้วเข้ารหัส base64url
 *   #z=<gzip base64url>   { v:1, issued:'yyyy-MM-dd', r:[record...] }
 * record: n ชื่อ, id รหัส, f = 1 ถ้ามีข้อมูลที่ฝึก (ถ้าไม่มี f พิมพ์เป็นฟอร์มเปล่าที่ใส่ชื่อนักศึกษา)
 *   c บริษัท, a ที่อยู่, p จังหวัด, ct โทรศัพท์สถานประกอบการ (v1.9.2), sn sp se st ผู้นิเทศงาน, co same|other, cn cp ce ที่ปรึกษาร่วม,
 *   pt ตำแหน่ง, jd ลักษณะงาน, p1 p2 โจทย์, wm onsite|hybrid|remote, s5 s4 ตารางเวลา, ip company|joint|student, cm ค่าตอบแทน
 * ไม่มีเบอร์โทรหรืออีเมลของนักศึกษาในข้อมูล (แบบตอบรับไม่ได้ใช้)
 * v1.9.2: โจทย์ ตารางเวลา และค่าตอบแทนส่งครบตามความยาวสูงสุดของช่อง หน้าแบบฟอร์ม (ฉบับ 1.2) ขึ้นบรรทัดใหม่และย่อตัวอักษรให้พอดี A4 เอง
 */

/** สถานะที่ถือว่าข้อมูลที่ฝึกผ่านการตรวจความครบถ้วนแล้ว พิมพ์ฉบับมีข้อมูลได้ */
var LETTER_FILLED_STATUSES_ = ['submitted', 'returned', 'agreement_sent', 'signed'];

/** ตัดข้อความยาวสำหรับแบบตอบรับ (รวมช่องว่าง ใส่ … ท้าย) แยกจาก clip_ ของ Logic.gs */
function letterClip_(s, n) {
  s = String(s === undefined || s === null ? '' : s).replace(/\s+/g, ' ').trim();
  return s.length > n ? s.slice(0, n - 1).trim() + '…' : s;
}

function letterRecord_(u, p) {
  var r = { n: String(u.name || u.email), id: String(u.student_id || '') };
  if (!p || LETTER_FILLED_STATUSES_.indexOf(String(p.status)) < 0) return r;
  r.f = 1;
  r.c = letterClip_(p.company_name, 120); r.a = letterClip_(p.company_address, 140); r.p = letterClip_(p.province, 40);
  r.ct = letterClip_(p.company_phone, 40);
  r.sn = letterClip_(p.supervisor_name, 80); r.sp = letterClip_(p.supervisor_position, 80);
  r.se = letterClip_(p.supervisor_email, 80); r.st = letterClip_(p.supervisor_phone, 30);
  r.co = p.coadvisor_same === 'no' ? 'other' : 'same';
  if (r.co === 'other') { r.cn = letterClip_(p.coadvisor_name, 60); r.cp = letterClip_(p.coadvisor_position, 40); r.ce = letterClip_(p.coadvisor_email, 60); }
  r.pt = letterClip_(p.position_title, 100); r.jd = letterClip_(p.job_description, 230);
  r.p1 = letterClip_(p.problem1_title, 200); r.p2 = letterClip_(p.problem2_title, 200);
  r.wm = String(p.work_mode || ''); r.s5 = letterClip_(p.schedule_305, 200); r.s4 = letterClip_(p.schedule_304, 200);
  r.ip = String(p.ip_owner || ''); r.cm = letterClip_(p.compensation, 200);
  return r;
}

/** ลิงก์หน้าแบบฟอร์มในพอร์ทัลพร้อมข้อมูล (ว่าง = ยังไม่ได้ตั้ง PORTAL_URL) */
function letterUrl_(records) {
  var base = portalUrl_('forms/intro-letter.html');
  if (!base) return '';
  var json = JSON.stringify({ v: 1, issued: todayStr_(), r: records });
  var gz = Utilities.gzip(Utilities.newBlob(json, 'application/json', 'letter.json'));
  return base + '#z=' + Utilities.base64EncodeWebSafe(gz.getBytes()).replace(/=+$/, '');
}

function letterOut_(records, actor, target) {
  var url = letterUrl_(records);
  if (!url) return { ok: false, error: 'ยังไม่ได้ตั้ง PORTAL_URL ในชีต Config' };
  audit_(actor, 'letter_link', target, records.length + ' record(s)');
  return { ok: true, url: url, count: records.length, filled: records.filter(function (r) { return r.f; }).length };
}

/** นักศึกษา: ลิงก์หนังสือแนะนำและแบบตอบรับของตนเอง (มีข้อมูลเมื่อส่งให้หลักสูตรตรวจแล้ว) */
function api_letterLink(session) {
  return safe_(function () {
    var u = requireUser_(session, ['student']);
    return letterOut_([letterRecord_(u, placementOf_(u.email))], u.email, u.email);
  });
}

/** อาจารย์นิเทศของนักศึกษาคนนั้นหรือผู้ดูแล: ลิงก์รายคน */
function api_adminLetterLink(session, email) {
  return safe_(function () {
    var me = requireUser_(session, ['teacher', 'admin']);
    var u = findUser_(normEmail_(email));
    if (!u || u.role !== 'student' || !canSeeStudent_(me, u)) return { ok: false, error: 'ไม่พบนักศึกษา หรือท่านไม่ได้เป็นอาจารย์นิเทศของนักศึกษาคนนี้' };
    return letterOut_([letterRecord_(u, placementOf_(u.email))], me.email, u.email);
  });
}

/** กลุ่มนักศึกษาสำหรับพิมพ์รวม which = filled | noreply | blank */
function letterGroups_() {
  var placements = {};
  readAll_('Placements').forEach(function (p) { placements[normEmail_(p.student_email)] = p; });
  var g = { filled: [], noreply: [], blank: [] };
  activeStudents_().slice().sort(function (a, b) { return (a.section + a.student_id).localeCompare(b.section + b.student_id); }).forEach(function (u) {
    var p = placements[u.email] || null;
    var rec = letterRecord_(u, p);
    if (rec.f) {
      g.filled.push(rec);
      if (!p.reply_file_id) g.noreply.push(rec);
    } else g.blank.push(rec);
  });
  return g;
}

/** ผู้ดูแล: ลิงก์พิมพ์รวมหลายคน ให้หัวหน้าหลักสูตรลงนามหน้า 1 ครั้งเดียว */
function api_adminLettersAll(session, which) {
  return safe_(function () {
    var me = requireUser_(session, ['admin']);
    var g = letterGroups_();
    if (!g[which]) return { ok: false, error: 'กลุ่มไม่ถูกต้อง' };
    if (!g[which].length) return { ok: false, error: 'ไม่มีนักศึกษาในกลุ่มนี้' };
    return letterOut_(g[which], me.email, 'all:' + which);
  });
}

/**
 * Logic.gs — ฟังก์ชันล้วน (ไม่เรียกบริการของ Google) ทดสอบได้ด้วย Node
 * ทุกฟังก์ชันลงท้ายด้วย _ เพื่อไม่ให้ฝั่งเบราว์เซอร์เรียกผ่าน google.script.run ได้
 */

var DAY_MS_ = 86400000;

function parseDate_(s) {
  var m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(s || '').trim());
  if (!m) return null;
  var y = +m[1], mo = +m[2] - 1, d = +m[3];
  var dt = new Date(Date.UTC(y, mo, d));
  if (dt.getUTCFullYear() !== y || dt.getUTCMonth() !== mo || dt.getUTCDate() !== d) return null;
  return dt;
}

function fmtDate_(d) { return d.toISOString().slice(0, 10); }

/** 'HH:mm' -> นาทีนับจากเที่ยงคืน */
function parseTime_(s) {
  var m = /^([01]?\d|2[0-3]):([0-5]\d)$/.exec(String(s || '').trim());
  return m ? (+m[1]) * 60 + (+m[2]) : null;
}

function mondayOf_(d) {
  var diff = (d.getUTCDay() + 6) % 7;
  return new Date(d.getTime() - diff * DAY_MS_);
}

/** สัปดาห์ที่ (จันทร์–อาทิตย์) นับจากสัปดาห์ที่มีวันเปิดภาค = 1 */
function weekIndex_(dateStr, semesterStartStr) {
  var d = parseDate_(dateStr), s = parseDate_(semesterStartStr);
  if (!d || !s) return null;
  return Math.floor((mondayOf_(d).getTime() - mondayOf_(s).getTime()) / (7 * DAY_MS_)) + 1;
}

function weekRange_(n, semesterStartStr) {
  var s = mondayOf_(parseDate_(semesterStartStr));
  var a = new Date(s.getTime() + (n - 1) * 7 * DAY_MS_);
  var b = new Date(a.getTime() + 6 * DAY_MS_);
  return { start: fmtDate_(a), end: fmtDate_(b) };
}

/** ชั่วโมงทำงาน = (เลิก − เริ่ม − พัก) ปัดทศนิยม 2 ตำแหน่ง หรือ null ถ้าไม่ถูกต้อง */
function computeHours_(start, end, breakMin) {
  var a = parseTime_(start), b = parseTime_(end);
  var br = Number(breakMin === '' || breakMin === undefined || breakMin === null ? 0 : breakMin);
  if (a === null || b === null || !(b > a) || isNaN(br) || br < 0) return null;
  var mins = b - a - br;
  if (mins <= 0) return null;
  return Math.round(mins / 60 * 100) / 100;
}

/** ลิงก์ Google Maps หรือพิกัด lat,lng */
function isMapRef_(s) {
  s = String(s || '').trim();
  if (/^-?\d{1,2}\.\d+\s*,\s*-?\d{1,3}\.\d+$/.test(s)) return true;
  return /^https:\/\/((www\.)?google\.[a-z.]+\/maps|maps\.google\.[a-z.]+|maps\.app\.goo\.gl|goo\.gl\/maps)/i.test(s);
}
/** ทำลิงก์เปิดแผนที่จากค่าที่กรอก */
function mapHref_(s) {
  s = String(s || '').trim();
  if (!s) return '';
  return /^https:/i.test(s) ? s : 'https://www.google.com/maps/search/?api=1&query=' + encodeURIComponent(s.replace(/\s+/g, ''));
}

var ABSENCE_REASONS = { sick: 'ป่วย', personal: 'ลากิจ', closed: 'บริษัทหยุดหรือวันหยุด', other: 'อื่นๆ' };

/** ลิงก์หลักฐาน: คั่นด้วยช่องว่างหรือขึ้นบรรทัดใหม่ ไม่เกิน 3 ลิงก์ ต้องขึ้นต้นด้วย http:// หรือ https:// */

function validateEntry_(e, ctx) {
  var errors = [];
  var d = parseDate_(e.date);
  if (!d) errors.push('วันที่ไม่ถูกต้อง');
  else {
    if (e.date < ctx.semesterStart) errors.push('วันที่อยู่ก่อนวันเปิดภาค (' + ctx.semesterStart + ')');
    if (e.date > ctx.semesterEnd) errors.push('วันที่เลยวันสิ้นสุดการฝึก (' + ctx.semesterEnd + ')');
    if (e.date > ctx.today) errors.push('บันทึกล่วงหน้าไม่ได้');
  }
  if (e.kind === 'absence') {
    if (!ABSENCE_REASONS[e.absence_reason]) errors.push('ต้องเลือกเหตุที่ไม่ได้ปฏิบัติงาน');
    if (e.absence_reason === 'other' && String(e.absence_note || '').trim().length < 3) errors.push('กรุณาระบุเหตุที่ไม่ได้ปฏิบัติงาน');
    if ((ctx.sameDay || []).some(function (o) { return o.entry_id !== e.entry_id && o.kind === 'absence'; })) errors.push('วันนี้บันทึกเหตุที่ไม่ได้ปฏิบัติงานไว้แล้ว');
    return { ok: errors.length === 0, errors: errors, hours: 0 };
  }
  if (e.course !== '305' && e.course !== '304') errors.push('ต้องเลือกรายวิชา 305 หรือ 304');
  if (!WORK_MODE_KEYS_[e.work_mode]) errors.push('ต้องเลือกรูปแบบการทำงานของวันนั้น');
  var act = String(e.activity || '').trim();
  if (act.length < MIN_ACTIVITY_CHARS) errors.push('รายละเอียดงานต้องยาวอย่างน้อย ' + MIN_ACTIVITY_CHARS + ' ตัวอักษร: บอกว่าทำอะไร ได้ผลอะไร ใช้เครื่องมืออะไร');
  if (act.length > 1000) errors.push('รายละเอียดงานยาวเกิน 1,000 ตัวอักษร');
  var links = splitLinks_(e.link_url);
  if (links.length > MAX_LINKS) errors.push('ใส่ลิงก์ได้ไม่เกิน ' + MAX_LINKS + ' ลิงก์');
  if (links.some(function (u) { return !/^https?:\/\/[^\s<>"]+$/i.test(u) || u.length > 300; })) errors.push('ลิงก์ต้องขึ้นต้นด้วย http:// หรือ https://');
  if (String(e.problems || '').length > 500) errors.push('ปัญหาและอุปสรรคยาวเกิน 500 ตัวอักษร');
  if (String(e.learned || '').length > 500) errors.push('สิ่งที่ได้เรียนรู้ยาวเกิน 500 ตัวอักษร');
  var hours = computeHours_(e.start, e.end, e.break_min);
  if (hours === null) errors.push('เวลาเริ่ม เวลาเลิก หรือเวลาพักไม่ถูกต้อง');
  else {
    var a = parseTime_(e.start), b = parseTime_(e.end);
    var total = hours;
    (ctx.sameDay || []).forEach(function (o) {
      if (o.entry_id === e.entry_id) return;
      total += Number(o.hours || 0);
      var oa = parseTime_(o.start), ob = parseTime_(o.end);
      if (oa !== null && ob !== null && a < ob && oa < b) errors.push('ช่วงเวลาซ้อนกับรายการ ' + o.start + '–' + o.end);
    });
    if (total > ctx.maxHoursPerDay) errors.push('รวมทั้งวันเกิน ' + ctx.maxHoursPerDay + ' ชั่วโมง');
  }
  return { ok: errors.length === 0, errors: errors, hours: hours };
}

/** สรุปชั่วโมงตามสถานะและรายวิชา */
function summarizeHours_(entries) {
  var z = function () { return { '305': 0, '304': 0 }; };
  var out = { certified: z(), pending: z(), draft: z(), returned: z() };
  entries.forEach(function (e) {
    var st = out[e.status] ? e.status : 'draft';
    var c = String(e.course);
    if (c !== '305' && c !== '304') return;
    out[st][c] = round2_(out[st][c] + Number(e.hours || 0));
  });
  return out;
}

/** ชั่วโมงที่ควรมี ณ วันนี้ (นับสัปดาห์ที่ผ่านไปครบแล้ว) */
function expectedHours_(todayStr, semesterStartStr, perWeek, target) {
  var w = weekIndex_(todayStr, semesterStartStr);
  if (w === null || w <= 1) return 0;
  return Math.min(target, (w - 1) * perWeek);
}

/** ค่าที่จะเขียนลงชีต: กันสูตร (= + - @) และกัน Sheets แปลงวันที่/ตัวเลขเอง */
function cellOut_(v) {
  if (v === null || v === undefined) return '';
  if (typeof v === 'number' || typeof v === 'boolean' || v instanceof Date) return v;
  var s = String(v);
  if (s === '') return '';
  if (/^[=+\-@]/.test(s) || /^[\d\s.:\/\-T+Z]+$/.test(s)) return "'" + s;
  return s;
}

function api_placementSave(session, data, submit) {
  return safe_(function () {
    var u = requireUser_(session, ['student']);
    data = data || {};
    return withLock_(function () {
      var p = placementOf_(u.email);
      if (p && p.status !== 'draft' && p.status !== 'returned') {
        return { ok: false, error: 'ส่งข้อมูลแล้ว แก้ไขไม่ได้ ถ้าต้องการเปลี่ยนกรุณาติดต่อผู้ประสานงาน SE-WIL' };
      }
      var clean = {};
      Object.keys(PLACEMENT_FIELDS_).forEach(function (k) { clean[k] = clip_(data[k], PLACEMENT_FIELDS_[k]); });
      clean.supervisor_email = normEmail_(clean.supervisor_email);
      clean.coadvisor_email = normEmail_(clean.coadvisor_email);
      if (clean.coadvisor_same === 'yes') {
        clean.coadvisor_name = ''; clean.coadvisor_position = ''; clean.coadvisor_email = ''; clean.coadvisor_phone = '';
      }
      if (clean.same_as_301 === 'yes') clean.change_reason = '';
      if (COMPANY_TYPES.indexOf(clean.company_type) < 0) clean.company_type = '';
      var consent = data.pdpa_consent === true || data.pdpa_consent === 'true';
      var cvRequired = String(cfg_().CV_REQUIRED).toUpperCase() === 'TRUE';
      var check = validatePlacement_(Object.assign({ pdpa_consent: consent, cv_required: cvRequired, has_cv: !!(p && p.cv_file_id) }, clean), !!submit);
      if (!check.ok) return { ok: false, errors: check.errors, error: check.errors.join('\n') };

      var now = nowIso_();
      clean.updated_at = now;
      if (consent && !(p && p.pdpa_consent_at)) clean.pdpa_consent_at = now;
      if (!consent) clean.pdpa_consent_at = '';
      var status = p ? p.status : 'draft';
      if (submit) {
        if (!canTransition_(status, 'submitted')) return { ok: false, error: 'สถานะปัจจุบันส่งไม่ได้' };
        clean.status = 'submitted';
        clean.submitted_at = now;
      }
      if (p) {
        // ชื่อและรหัสตามชีต Users ล่าสุด (ผู้ดูแลอาจเติมชื่อหลังนักศึกษาบันทึกร่างครั้งแรก)
        if (u.name) clean.student_name = String(u.name);
        if (u.student_id) clean.student_id = String(u.student_id);
        update_('Placements', p._row, clean);
      } else {
        clean.placement_id = shortId_('PL', 8);
        clean.student_email = u.email;
        clean.student_id = String(u.student_id || '');
        clean.student_name = String(u.name || '');
        clean.status = clean.status || 'draft';
        clean.created_at = now;
        insert_('Placements', clean);
      }
      audit_(u.email, submit ? 'placement_submit' : 'placement_save', '', '');
      return { ok: true, status: clean.status || status };
    });
  });
}

function api_adminOverview(session) {
  return safe_(function () {
    var me = requireUser_(session, ['teacher', 'admin']);
    var sem = semesterPayload_();
    var students = readAll_('Users').filter(function (u) {
      return String(u.role).trim().toLowerCase() === 'student' && String(u.active).toUpperCase() !== 'FALSE' &&
        (me.role === 'admin' || normEmail_(u.advisor_email) === me.email);
    });
    var placements = {};
    readAll_('Placements').forEach(function (p) { placements[normEmail_(p.student_email)] = p; });
    var byStudent = {};
    readAll_('Logbook').forEach(function (e) {
      var k = normEmail_(e.student_email);
      (byStudent[k] = byStudent[k] || []).push(e);
    });
    var exp305 = expectedHours_(sem.today, sem.start, sem.weekly305, sem.target305);
    var exp304 = expectedHours_(sem.today, sem.start, sem.weekly304, sem.target304);
    var counts = { none: 0, draft: 0, submitted: 0, returned: 0, agreement_sent: 0, signed: 0 };
    var rows = students.map(function (u) {
      var email = normEmail_(u.email);
      var p = placements[email];
      var entries = byStudent[email] || [];
      var sum = summarizeHours_(entries);
      var last = entries.reduce(function (m, e) { return e.date > m ? e.date : m; }, '');
      var status = p ? p.status : 'none';
      counts[status] = (counts[status] || 0) + 1;
      var flags = [];
      var open = p && LOGBOOK_OPEN_STATUSES.indexOf(p.status) >= 0;
      if (open && exp305 > 0 && sum.certified['305'] + sum.pending['305'] < 0.8 * exp305) flags.push('ชั่วโมง 305 ต่ำกว่าเป้า');
      if (open && exp304 > 0 && sum.certified['304'] + sum.pending['304'] < 0.8 * exp304) flags.push('ชั่วโมง 304 ต่ำกว่าเป้า');
      if (sum.returned['305'] + sum.returned['304'] > 0) flags.push('มีรายการถูกส่งกลับ');
      return {
        email: email, name: String(u.name || ''), student_id: String(u.student_id || ''), section: String(u.section || ''),
        advisor_email: normEmail_(u.advisor_email), status: status,
        placement_id: p ? p.placement_id : '', company_name: p ? p.company_name : '',
        supervisor_name: p ? p.supervisor_name : '', has_reply: !!(p && p.reply_file_id), reply_uploaded_at: p ? String(p.reply_uploaded_at || '') : '',
        hours: sum, last_entry: last, flags: flags
      };
    });
    return {
      ok: true, me: profile_(me), scope: me.role === 'admin' ? 'all' : 'advisees', semester: sem, expected: { '305': exp305, '304': exp304 },
      counts: counts, rows: rows, todo: staffTodo_(me, rows), selfAssign: selfAssignOpen_(), options: optionsPayload_()
    };
  });
}

/**
 * เลขข้อตกลงถัดไป = เลขสูงสุดที่เคยออกแล้ว + 1 (v1.9.1)
 * เดิมนับจำนวนแถวที่มีเลข ถ้าแถวถูกลบ (รีเซ็ตนักศึกษาหรือลบในชีต) จะได้เลขซ้ำกับที่ออกไปแล้ว
 * จึงดูทั้งชีต Placements, Audit (placement_approve) และ Trash ที่เก็บแถวที่ถูกลบ เลขที่ออกแล้วไม่ถูกนำมาใช้ใหม่
 */
function nextAgreementNo_() {
  var c = cfg_();
  var year = String(c.SEMESTER_LABEL || '').split('/').pop() || '';
  var prefix = 'SE-WIL01/' + year + '-';
  var re = new RegExp(prefix.replace(/[.*+?^${}()|[\]\\\/-]/g, '\\$&') + '(\\d+)', 'g');
  var max = 0;
  function scan(s) {
    String(s || '').replace(re, function (m, d) { var n = parseInt(d, 10); if (n > max) max = n; return m; });
  }
  readAll_('Placements').forEach(function (r) { scan(r.agreement_no); });
  try { readAll_('Audit').forEach(function (r) { if (r.action === 'placement_approve') scan(r.detail); }); } catch (e) { }
  try { readAll_('Trash').forEach(function (r) { if (r.sheet === 'Placements') scan(r.row_json); }); } catch (e) { }
  return prefix + ('000' + (max + 1)).slice(-3);
}

/** เก็บสำเนาแถวไว้ในชีต Trash ก่อนลบหรือแก้ */

function runReset_(plan, actor, opts) {

function ensureSheet_(ss, name, headers) {
  var sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  var current = sh.getLastColumn() > 0 ? sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String) : [];
  var missing = headers.filter(function (h) { return current.indexOf(h) < 0; });
  if (current.length === 0 || current.join('') === '') {
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  } else if (missing.length) {
    sh.getRange(1, current.length + 1, 1, missing.length).setValues([missing]);
  }
  sh.setFrozenRows(1);
  try { sh.getRange(1, 1, 1, sh.getLastColumn()).setFontWeight('bold'); } catch (e) { }
  return sh;
}

/**
 * Todo.gs — "สิ่งที่ต้องทำ" และ "กำหนดการถัดไป" ของแต่ละบทบาท
 * kind: do = ต้องทำ, late = เลยกำหนด, wait = รอฝ่ายอื่น, ok = เรียบร้อย
 * go: แท็บที่ปุ่ม "ไปทำ" พาไป (place, log, assign, docs, overview)
 */

function todoItem_(kind, text, opt) {
  var o = { kind: kind, text: text };
  if (opt) Object.keys(opt).forEach(function (k) { if (opt[k] !== undefined && opt[k] !== '') o[k] = opt[k]; });
  return o;
}

function addDays_(iso, n) {
  var d = parseDate_(iso);
  return d ? fmtDate_(new Date(d.getTime() + n * DAY_MS_)) : iso;
}

/** กำหนดการของบทบาท (admin เห็นของ teacher ด้วย) วันที่จาก Config ถ้ารายการระบุ cfg */
function planFor_(role) {
  var c = cfg_();
  return TERM_PLAN.filter(function (x) {
    return x.roles.indexOf(role) >= 0 || (role === 'admin' && x.roles.indexOf('teacher') >= 0);
  }).map(function (x) {
    return { date: (x.cfg && /^\d{4}-\d{2}-\d{2}$/.test(String(c[x.cfg] || ''))) ? String(c[x.cfg]) : x.date, title: x.title };
  });
}

function upcoming_(role, today, n) {
  return planFor_(role).filter(function (x) { return x.date >= today; })
    .sort(function (a, b) { return a.date < b.date ? -1 : (a.date > b.date ? 1 : 0); })
    .slice(0, n || 4);
}

function approvalDue_() {
  var it = TERM_PLAN.filter(function (x) { return x.roles.indexOf('admin') >= 0 && /อนุมัติ/.test(x.title); })[0];
  return it ? it.date : '';
}

var ENTRY_FIELDS_ = ['date', 'start', 'end', 'break_min', 'course', 'work_mode', 'activity', 'kind', 'absence_reason', 'absence_note', 'link_url', 'problems', 'learned'];
/** รูปหลักฐานใน logbook: 1 รูปต่อรายการ เก็บในโฟลเดอร์ของรายวิชา เปิดได้เฉพาะนักศึกษา อาจารย์นิเทศ ผู้ดูแล และผู้นิเทศงาน */
var LOG_PHOTO_SPEC_ = { folderKey: 'LOG_PHOTO_FOLDER_ID', folderName: 'รูป logbook' };
var LOG_PHOTO_MIMES_ = { 'image/jpeg': 'jpg', 'image/png': 'png', 'image/webp': 'webp' };

function openPlacementOf_(email) {
  var p = placementOf_(email);
  if (!p) throw userError_('ยังไม่มีข้อมูลที่ฝึก กรุณากรอกและส่งข้อมูลที่ฝึกก่อน');
  if (LOGBOOK_OPEN_STATUSES.indexOf(p.status) < 0) throw userError_('บันทึก logbook ได้หลังหลักสูตรอนุมัติข้อมูลที่ฝึกแล้ว');
  return p;
}

function entriesOf_(email) {
  email = normEmail_(email);
  return readAll_('Logbook').filter(function (e) { return normEmail_(e.student_email) === email; });
}

function entryOut_(e) {
  var o = publicRow_(e, ['photo_file_id']);
  o.hours = Number(o.hours || 0);
  o.course = String(o.course);
  o.kind = e.kind === 'absence' ? 'absence' : 'work';
  o.has_photo = !!e.photo_file_id;
  return o;
}

function api_logList(session) {
  return safe_(function () {
    var u = requireUser_(session, ['student']);
    var p = placementOf_(u.email);
    var entries = entriesOf_(u.email).sort(function (a, b) {
      return a.date === b.date ? String(a.start).localeCompare(String(b.start)) : (a.date < b.date ? 1 : -1);
    });
    var sem = semesterPayload_();
    var batches = readAll_('CertBatches').filter(function (b) { return normEmail_(b.student_email) === u.email; });
    var lastManual = batches.reduce(function (m, b) { return String(b.created_at) > m ? String(b.created_at) : m; }, '');
    return {
      ok: true, me: profile_(u), semester: sem,
      open: !!p && LOGBOOK_OPEN_STATUSES.indexOf(p.status) >= 0,
      placement: p ? { status: p.status, company_name: p.company_name, supervisor_name: p.supervisor_name, work_mode: p.work_mode } : null,
      entries: entries.map(entryOut_), summary: summarizeHours_(entries),
      expected: {
        '305': expectedHours_(sem.today, sem.start, sem.weekly305, sem.target305),
        '304': expectedHours_(sem.today, sem.start, sem.weekly304, sem.target304)
      },
      lastBatchAt: lastManual
    };
  });
}

function api_logSave(session, input) {
  return safe_(function () {
    var u = requireUser_(session, ['student']);
    input = input || {};
    return withLock_(function () {
      var p = openPlacementOf_(u.email);
      var all = entriesOf_(u.email);
      var existing = null;
      if (input.entry_id) {
        existing = all.filter(function (e) { return e.entry_id === input.entry_id; })[0];
        if (!existing) return { ok: false, error: 'ไม่พบรายการ' };
        if (existing.status !== 'draft' && existing.status !== 'returned') return { ok: false, error: 'รายการที่ส่งรับรองแล้วแก้ไขไม่ได้' };
      }
      var absence = input.kind === 'absence';
      var e = {
        entry_id: existing ? existing.entry_id : '', kind: absence ? 'absence' : 'work',
        date: clip_(input.date, 10), start: absence ? '' : clip_(input.start, 5), end: absence ? '' : clip_(input.end, 5),
        break_min: absence ? 0 : Number(input.break_min || 0), course: absence ? '' : clip_(input.course, 3),
        work_mode: absence ? '' : clip_(input.work_mode, 10), activity: absence ? '' : clip_(input.activity, 1000),
        absence_reason: absence ? clip_(input.absence_reason, 10) : '', absence_note: absence ? clip_(input.absence_note, 200) : '',
        link_url: absence ? '' : splitLinks_(clip_(input.link_url, 1000)).join(' '),
        problems: absence ? '' : clip_(input.problems, 600), learned: absence ? '' : clip_(input.learned, 600)
      };
      var c = cfg_();
      var check = validateEntry_(e, {
        semesterStart: c.SEMESTER_START, semesterEnd: c.SEMESTER_END, today: todayStr_(),
        maxHoursPerDay: cfgNum_('MAX_HOURS_PER_DAY') || 12,
        sameDay: all.filter(function (o) { return o.date === e.date; }).map(function (o) { return Object.assign({}, o, { kind: o.kind === 'absence' ? 'absence' : 'work' }); })
      });
      if (!check.ok) return { ok: false, errors: check.errors, error: check.errors.join('\n') };
      ensureColumns_('Logbook');
      var now = nowIso_();
      var fields = {
        date: e.date, start: e.start, end: e.end, break_min: e.break_min, hours: check.hours, course: e.course,
        work_mode: e.work_mode, activity: e.activity, kind: e.kind, absence_reason: e.absence_reason, absence_note: e.absence_note,
        link_url: e.link_url, problems: e.problems, learned: e.learned, status: 'draft', batch_id: '', return_comment: '', updated_at: now
      };
      if (existing) {
        if (absence && existing.photo_file_id) {
          try { DriveApp.getFileById(existing.photo_file_id).setTrashed(true); } catch (x) { /* ไม่มีไฟล์แล้ว */ }
          fields.photo_file_id = '';
        }
        update_('Logbook', existing._row, fields);
      } else {
        e.entry_id = shortId_('E', 10);
        insert_('Logbook', Object.assign({ entry_id: e.entry_id, placement_id: p.placement_id, student_email: u.email, created_at: now, photo_file_id: '' }, fields));
      }
      return { ok: true, entry_id: e.entry_id, hours: check.hours, kind: e.kind };
    });
  });
}

function ownEditableEntry_(u, entryId) {
  var e = findOne_('Logbook', function (r) { return r.entry_id === entryId && normEmail_(r.student_email) === u.email; });
  if (!e) throw userError_('ไม่พบรายการ');

/** นักศึกษาแนบหรือเปลี่ยนรูป 1 รูปต่อรายการ (หน้าเว็บย่อรูปก่อนส่ง) */
function api_logPhotoUpload(session, entryId, base64) {
  return safe_(function () {
    var u = requireUser_(session, ['student']);
    var maxMb = cfgNum_('MAX_UPLOAD_MB') || 5;
    base64 = String(base64 || '').replace(/^data:[^,]*,/, '');
    if (!base64) return { ok: false, error: 'ไม่พบไฟล์' };
    if (base64.length * 0.75 > maxMb * 1048576 + 4) return { ok: false, error: 'รูปใหญ่เกิน ' + maxMb + ' MB' };
    var bytes;
    try { bytes = Utilities.base64Decode(base64); } catch (x) { return { ok: false, error: 'อ่านไฟล์ไม่ได้' }; }
    var mime = sniffMime_(bytes);
    if (!LOG_PHOTO_MIMES_[mime]) return { ok: false, error: 'รูปต้องเป็นไฟล์ JPG PNG หรือ WebP' };
    return withLock_(function () {
      var e = ownEditableEntry_(u, String(entryId || ''));
      ensureColumns_('Logbook');
      var name = safeFileName_(String(u.student_id || 'student') + '_' + e.date + '_' + e.entry_id) + '.' + LOG_PHOTO_MIMES_[mime];
      var file = uploadFolder_(LOG_PHOTO_SPEC_).createFile(Utilities.newBlob(bytes, mime, name));
      if (e.photo_file_id) { try { DriveApp.getFileById(e.photo_file_id).setTrashed(true); } catch (x) { /* ไม่มีไฟล์แล้ว */ } }
      update_('Logbook', e._row, { photo_file_id: file.getId(), updated_at: nowIso_() });
      audit_(u.email, 'log_photo', e.entry_id, bytes.length + ' bytes');
      return { ok: true };
    });
  });
}

function api_logSendNow(session) {
  return safe_(function () {
    var u = requireUser_(session, ['student']);
    return withLock_(function () {
      var p = openPlacementOf_(u.email);
      var dayAgo = new Date(Date.now() - 20 * 3600000).toISOString();
      var recent = readAll_('CertBatches').filter(function (b) {
        return normEmail_(b.student_email) === u.email && String(b.created_at) > dayAgo;
      });
      if (recent.length) return { ok: false, error: 'ส่งรับรองได้วันละครั้ง ระบบจะส่งรอบปกติทุกวัน' + dayTh_(cfg_().DIGEST_DAY) };
      var b = createBatchFor_(p, todayStr_());
      if (!b) return { ok: false, error: 'ไม่มีรายการร่างที่รอส่ง' };
      notifySupervisors_([normEmail_(p.supervisor_email)]);
      audit_(u.email, 'batch_send_now', b.batch_id, '');
      return { ok: true, batch_id: b.batch_id };
    });
  });
}

/* ---------- การสร้างชุดรับรองและสรุปรายสัปดาห์ ---------- */

/** รวมรายการร่างที่วันที่ไม่เกิน upToDate ของ placement เป็นชุดรับรองหนึ่งชุด */
function createBatchFor_(p, upToDate) {
  var email = normEmail_(p.student_email);
  var drafts = readAll_('Logbook').filter(function (e) {
    return normEmail_(e.student_email) === email && e.status === 'draft' && e.date <= upToDate;
  });
  if (!drafts.length) return null;
  var sum = summarizeHours_(drafts.map(function (e) { return { status: 'pending', course: e.course, hours: e.hours }; }));
  var batch = {
    batch_id: shortId_('B', 8), placement_id: p.placement_id, student_email: email,
    supervisor_email: normEmail_(p.supervisor_email), created_at: nowIso_(),
    entry_ids: drafts.map(function (e) { return e.entry_id; }).join(','),
    hours_305: sum.pending['305'], hours_304: sum.pending['304'], status: 'pending',
    decided_at: '', decided_by_name: '', comment: ''
  };
  insert_('CertBatches', batch);
  drafts.forEach(function (e) { update_('Logbook', e._row, { status: 'pending', batch_id: batch.batch_id, updated_at: nowIso_() }); });
  return batch;
}

/** ส่งอีเมลสรุปถึงผู้นิเทศงานที่มีชุดรอรับรอง (เฉพาะรายชื่อที่ให้มา หรือทั้งหมดถ้าไม่ระบุ) */
function notifySupervisors_(onlyEmails) {
  var pending = readAll_('CertBatches').filter(function (b) { return b.status === 'pending'; });
  var absentIds = {};
  readAll_('Logbook').forEach(function (e) { if (e.kind === 'absence') absentIds[e.entry_id] = true; });
  var placements = {};
  readAll_('Placements').forEach(function (p) { placements[p.placement_id] = p; });
  var bySv = {};
  pending.forEach(function (b) {
    var p = placements[b.placement_id];
    if (!p) return;
    var sv = normEmail_(p.supervisor_email);
    if (onlyEmails && onlyEmails.indexOf(sv) < 0) return;
    var key = normEmail_(b.student_email);
    bySv[sv] = bySv[sv] || {};
    var it = bySv[sv][key] = bySv[sv][key] || { student_name: p.student_name, student_id: p.student_id, hours_305: 0, hours_304: 0, count: 0, absences: 0 };
    it.hours_305 = round2_(it.hours_305 + Number(b.hours_305 || 0));
    it.hours_304 = round2_(it.hours_304 + Number(b.hours_304 || 0));
    String(b.entry_ids || '').split(',').filter(String).forEach(function (id) { if (absentIds[id]) it.absences++; else it.count++; });
  });
  var sent = 0;
  Object.keys(bySv).forEach(function (svEmail) {
    var anyP = null;
    Object.keys(placements).some(function (id) { if (normEmail_(placements[id].supervisor_email) === svEmail) { anyP = placements[id]; return true; } return false; });
    var sv = getOrCreateSupervisor_(svEmail, anyP ? anyP.supervisor_name : '');
    var items = Object.keys(bySv[svEmail]).map(function (k) { return bySv[svEmail][k]; });
    try { sendDigestEmail_(sv, items); sent++; } catch (e) { audit_('system', 'digest_mail_failed', svEmail, e.message); }
  });
  return sent;
}

function runDigest_() {
  return withLock_(function () {
    var today = todayStr_();
    var created = 0;
    readAll_('Placements').forEach(function (p) {
      if (LOGBOOK_OPEN_STATUSES.indexOf(p.status) < 0) return;
      if (createBatchFor_(p, today)) created++;
    });
    var sent = notifySupervisors_(null);
    cleanupSessions_();
    PropertiesService.getScriptProperties().setProperty('LAST_DIGEST_AT', nowIso_());
    audit_('system', 'digest', '', 'batches=' + created + ' emails=' + sent);
    return { batches: created, emails: sent };
  });
}

/** ตัวจัดเวลารายสัปดาห์เรียกฟังก์ชันนี้ (กันการเรียกถี่: ห่างกันอย่างน้อย 30 นาที) */
function job_weeklyDigest() {
  var active = '';
  try { active = Session.getActiveUser().getEmail(); } catch (e) { }
  var eff = '';
  try { eff = Session.getEffectiveUser().getEmail(); } catch (e) { }
  if (active && active !== eff) throw new Error('ไม่อนุญาต');
  var last = PropertiesService.getScriptProperties().getProperty('LAST_DIGEST_AT');
  if (last && Date.now() - Date.parse(last) < 30 * 60000) return { skipped: true };
  return runDigest_();
}

function api_adminRunDigest(session) {
  return safe_(function () {

}

function cleanupSessions_() {
  var cutoff = Date.now() - 30 * DAY_MS_;
  var rows = readAll_('Sessions').filter(function (s) { return Date.parse(s.expires_at) < cutoff; });
  rows.sort(function (a, b) { return b._row - a._row; }).forEach(function (s) { deleteRow_('Sessions', s._row); });
}

/* ---------- ผู้นิเทศงาน ---------- */

function api_svGet(k) {
  return safe_(function () {
    var sv = requireSupervisor_(k);
    var mine = readAll_('Placements').filter(function (p) {
      return normEmail_(p.supervisor_email) === sv.email && ['agreement_sent', 'signed'].indexOf(p.status) >= 0;
    });
    var coOf = readAll_('Placements').filter(function (p) {
      return p.coadvisor_same !== 'yes' && normEmail_(p.coadvisor_email) === sv.email && ['agreement_sent', 'signed'].indexOf(p.status) >= 0;
    });
    var ids = mine.map(function (p) { return p.placement_id; });
    var batches = readAll_('CertBatches').filter(function (b) { return ids.indexOf(b.placement_id) >= 0; });
    var entries = {};
    readAll_('Logbook').forEach(function (e) { if (ids.indexOf(e.placement_id) >= 0) entries[e.entry_id] = entryOut_(e); });
    var pending = batches.filter(function (b) { return b.status === 'pending'; }).map(function (b) {
      var o = publicRow_(b);
      o.entries = String(b.entry_ids || '').split(',').filter(String).map(function (id) { return entries[id]; }).filter(Boolean)
        .sort(function (a, c) { return a.date === c.date ? String(a.start).localeCompare(String(c.start)) : (a.date < c.date ? -1 : 1); });
      var p = mine.filter(function (x) { return x.placement_id === b.placement_id; })[0];
      o.student_name = p ? p.student_name : ''; o.student_id = p ? p.student_id : '';
      return o;
    });
    var recent = batches.filter(function (b) { return b.status !== 'pending'; })
      .sort(function (a, b) { return String(b.decided_at).localeCompare(String(a.decided_at)); }).slice(0, 10)
      .map(function (b) {
        var o = publicRow_(b);
        var p = mine.filter(function (x) { return x.placement_id === b.placement_id; })[0];
        o.student_name = p ? p.student_name : '';
        return o;
      });
    var totals = mine.map(function (p) {
      var es = Object.keys(entries).map(function (id) { return entries[id]; }).filter(function (e) { return e.placement_id === p.placement_id; });
      var s = summarizeHours_(es);
      return { placement_id: p.placement_id, student_name: p.student_name, certified: s.certified, pending: s.pending };
    });
    return {
      ok: true, supervisor: { name: sv.name, email: sv.email }, coordinator: coordinator_(), head: programHead_(),
      agreements: mine.map(placementForSupervisor_), coadvisorOf: coOf.map(placementForSupervisor_),
      pending: pending, recent: recent, totals: totals, semester: semesterPayload_(), dayTh: dayTh_(cfg_().DIGEST_DAY),
      todo: supervisorTodo_(mine, pending)
    };
  });
}

/** ผู้นิเทศงานรับรองหรือส่งกลับรายการในชุด certifyIds + returnIds ต้องครบทุกรายการในชุด */
function api_svDecide(k, batchId, certifyIds, returnIds, comment) {
  return safe_(function () {
    var sv = requireSupervisor_(k);
    certifyIds = (certifyIds || []).map(String);
    returnIds = (returnIds || []).map(String);
    comment = clip_(comment, 600);
    if (returnIds.length && comment.length < 5) return { ok: false, error: 'กรุณาระบุเหตุผลที่ส่งกลับให้นักศึกษาแก้ไข' };
    return withLock_(function () {
      var b = findOne_('CertBatches', function (r) { return r.batch_id === batchId; });
      if (!b) return { ok: false, error: 'ไม่พบชุดรายการ' };
      var p = placementById_(b.placement_id);
      if (!p || normEmail_(p.supervisor_email) !== sv.email) return { ok: false, error: 'รายการนี้ไม่ได้อยู่ในความรับผิดชอบของท่าน' };
      if (b.status !== 'pending') return { ok: false, error: 'ชุดนี้ได้รับการพิจารณาแล้ว' };
      var ids = String(b.entry_ids || '').split(',').filter(String);
      var all = certifyIds.concat(returnIds);
      var overlap = certifyIds.some(function (id) { return returnIds.indexOf(id) >= 0; });
      var complete = ids.every(function (id) { return all.indexOf(id) >= 0; }) && all.every(function (id) { return ids.indexOf(id) >= 0; });
      if (overlap || !complete) return { ok: false, error: 'กรุณาเลือกให้ครบทุกรายการ (รับรองหรือส่งกลับ)' };
      var now = nowIso_();
      readAll_('Logbook').forEach(function (e) {
        if (ids.indexOf(e.entry_id) < 0) return;
        if (certifyIds.indexOf(e.entry_id) >= 0) update_('Logbook', e._row, { status: 'certified', updated_at: now });
        else update_('Logbook', e._row, { status: 'returned', return_comment: comment, updated_at: now });
      });
      var status = returnIds.length === 0 ? 'certified' : (certifyIds.length === 0 ? 'returned' : 'partial');
      update_('CertBatches', b._row, { status: status, decided_at: now, decided_by_name: sv.name || sv.email, comment: comment });
      if (returnIds.length) {
        try { sendEntriesReturnedEmail_(p.student_email, p.student_name, sv.name || sv.email, returnIds.length, comment); }
        catch (e) { audit_('system', 'return_mail_failed', p.student_email, e.message); }
      }
      audit_(sv.email, 'batch_' + status, b.batch_id, 'certified=' + certifyIds.length + ' returned=' + returnIds.length);
      return { ok: true, status: status };
    });
  });

(function () {
  var C = window.SEWIL || {}; var TH = ['มกราคม','กุมภาพันธ์','มีนาคม','เมษายน','พฤษภาคม','มิถุนายน','กรกฎาคม','สิงหาคม','กันยายน','ตุลาคม','พฤศจิกายน','ธันวาคม'];
  function th(iso) { var d = iso && new Date(iso + 'T00:00:00'); return d && !isNaN(d) ? d.getDate() + ' ' + TH[d.getMonth()] + ' ' + (d.getFullYear() + 543) : ''; }
  function setAll(sel, v) { if (!v) return; document.querySelectorAll(sel).forEach(function (e) { e.textContent = v; }); }
  function fillConfig() {
    var head = (C.contacts || []).filter(function (c) { return /หัวหน้าหลักสูตร/.test(c.role) && c.name; })[0];
    var pc = (C.contacts || []).filter(function (c) { return c.primary; })[0];
    var t = C.term || {};
    setAll('[data-cfg="headName"]', head && head.name);
    setAll('[data-contact-primary]', pc && (pc.name + (pc.email ? ' (' + pc.email + ')' : '')));

    if (t.start && t.end) setAll('[data-cfg="period"]', th(t.start) + ' ถึง ' + th(t.end));
    var ad = (C.deadlines || []).filter(function (d) { return /SE-WIL01/.test(d.title) && /ยืนยัน/.test(d.title); })[0];
    if (ad) setAll('[data-cfg="agreementDue"]', th(ad.date));
    var now = new Date(); setAll('[data-letter-date]', now.getDate() + ' ' + TH[now.getMonth()] + ' ' + (now.getFullYear() + 543));
  }

  /* ---------- ฉบับมีข้อมูลจากระบบ SE-WIL (ข้อมูลอยู่ในเบราว์เซอร์นี้เท่านั้น) ---------- */
  var KEY = 'sewil-intro-letter';
  function store(v) { try { if (v === null) sessionStorage.removeItem(KEY); else if (v !== undefined) sessionStorage.setItem(KEY, v); return sessionStorage.getItem(KEY); } catch (e) { return null; } }
  function b64(s) {
    s = s.replace(/-/g, '+').replace(/_/g, '/'); while (s.length % 4) s += '=';
    var bin = atob(s), out = new Uint8Array(bin.length);
    for (var i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
    return out;
  }
  function readHash() {
    var m = /^#([zd])=([A-Za-z0-9_-]+)$/.exec(location.hash || '');
    if (!m) return Promise.resolve(store());
    try { history.replaceState(null, '', location.pathname + location.search); } catch (e) { }
    var bytes = b64(m[2]);
    if (m[1] === 'd') return Promise.resolve(store(new TextDecoder().decode(bytes)));
    if (typeof DecompressionStream === 'undefined') return Promise.reject(new Error('nostream'));
    return new Response(new Blob([bytes]).stream().pipeThrough(new DecompressionStream('gzip'))).text().then(function (t) { return store(t); });
  }
  function el(tag, text, cls) { var e = document.createElement(tag); if (text) e.textContent = text; if (cls) e.className = cls; return e; }
  function put(root, f, v) {
    if (v === undefined || v === null || v === '') return;
    root.querySelectorAll('[data-f="' + f + '"]').forEach(function (e) { e.textContent = v; e.classList.add('fv'); });
  }
  function tick(root, k) { var b = root.querySelector('[data-box="' + k + '"]'); if (b) b.classList.add('on'); }
  function fill(r, p1, p2) {

    cl.appendChild(el('b', r.n));
    cl.appendChild(document.createTextNode((r.id ? ' รหัสนักศึกษา ' + r.id : '') + ' เข้าฝึก' +
      (r.f ? ' และขอความกรุณาตรวจข้อมูลในแบบตอบรับ (หน้าถัดไป) ลงนาม แล้วมอบให้นักศึกษา จักขอบคุณยิ่ง'
           : ' หากรับนักศึกษาได้ ขอความกรุณากรอกแบบตอบรับเบื้องต้น (หน้าถัดไป) ลงนาม แล้วมอบให้นักศึกษา จักขอบคุณยิ่ง')));
    put(p2, 'student', who);
    if (!r.f) return;
    put(p1, 'to', [r.sn, r.c].filter(Boolean).join(' '));
    p2.querySelector('[data-f="note"]').textContent = 'ข้อมูลจากที่นักศึกษากรอกในระบบ SE-WIL โปรดตรวจ ถ้าไม่ถูกต้องให้แก้ด้วยปากกาและลงนามกำกับ แล้วลงนามด้านล่าง';
    p2.classList.add('filled');
    put(p2, 'company', r.c); put(p2, 'address', [r.a, r.p ? 'จ.' + r.p : ''].filter(Boolean).join(' '));
    put(p2, 'ct', r.ct); /* ระบบรุ่นก่อน v1.9.2 ไม่ส่งเบอร์สถานประกอบการ ช่องนี้เว้นเส้นประให้เขียนเอง */
    tick(p2, 'accept');
    put(p2, 'sn', r.sn); put(p2, 'sp', r.sp); put(p2, 'se', r.se); put(p2, 'st', r.st || '-');
    if (r.co === 'other') { tick(p2, 'co-other'); put(p2, 'co', [r.cn, r.cp, r.ce].filter(Boolean).join(' / ')); } else { tick(p2, 'co-same'); put(p2, 'co', '-'); }
    put(p2, 'pt', r.pt); put(p2, 'jd', r.jd); put(p2, 'p1', r.p1); put(p2, 'p2', r.p2 || '-');
    if (r.wm) tick(p2, 'wm-' + r.wm);
    put(p2, 's5', r.s5); put(p2, 's4', r.s4);
    if (r.ip) tick(p2, 'ip-' + r.ip);
    if (r.cm && !/^ไม่มี/.test(r.cm)) { tick(p2, 'cm-yes'); put(p2, 'cm', r.cm); } else { tick(p2, 'cm-none'); put(p2, 'cm', '-'); }

     หน้าจอกับตอนพิมพ์ใช้ขนาดเดียวกัน ถ้าเนื้อหาของหน้าใดยาวเกินขอบล่าง (ข้อมูลยาว) ย่อตัวอักษรของหน้านั้นทีละน้อย

  var MIN_FS = 10.5;
  function contentBottom(pg) {
    var m = 0;
    Array.prototype.forEach.call(pg.children, function (c) { if (!c.classList.contains('foot')) m = Math.max(m, c.getBoundingClientRect().bottom); });
    return m;
  }
  function fitPage(pg) {
    pg.style.fontSize = '';
    if (window.matchMedia && matchMedia('screen and (max-width:800px)').matches) return; // จอมือถือ: หน้าไม่ล็อกความสูง
    var fs = parseFloat(getComputedStyle(pg).fontSize), n = 0;
    function limit() { return pg.getBoundingClientRect().top + pg.clientHeight - parseFloat(getComputedStyle(pg).paddingBottom); }
    while (contentBottom(pg) > limit() + 0.5 && fs > MIN_FS && n++ < 40) { fs -= 0.25; pg.style.fontSize = fs + 'px'; }
    pg.setAttribute('data-fs', fs);
  }
  function fitAll() { document.querySelectorAll('section.page').forEach(fitPage); }
  window.addEventListener('resize', fitAll);
  window.addEventListener('beforeprint', fitAll);
  fillConfig();
  readHash().then(function (t) {
    if (!t) return;
    var data = null; try { data = JSON.parse(t); } catch (e) { }

  }).catch(function () {

  }).then(function () {
    fitAll();
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(fitAll);
  });
})();
