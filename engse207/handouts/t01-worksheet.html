web/src/components/TaskForm.jsxหน้าจอ (React)
export default function TaskForm({ onCreated }) {
  const [title, setTitle] = useState("");
  const [error, setError] = useState("");

  async function handleSubmit(event) {
    event.preventDefault();
    setError("");
    try {
      const task = await createTask({ title });
      onCreated(task);              // เพิ่มงานใหม่เข้ารายการ
      setTitle("");
    } catch (err) {
      setError(err.message);        // แสดงข้อความใต้ฟอร์ม
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <input value={title} onChange={(e) => setTitle(e.target.value)} />
      <button type="submit">สร้างงาน</button>
      {error && <p className="error">{error}</p>}
    </form>
  );
}
web/src/api/tasks.jsหน้าจอ → เรียก API
export async function createTask(data) {
  const res = await fetch("/api/tasks", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(data),
  });
  const body = await res.json();
  if (!res.ok) throw new Error(body.error);
  return body;
}
api/src/routes/tasks.routes.jsAPI · รับคำขอ
router.post("/tasks", async (req, res) => {
  try {
    const task = await taskService.createTask(req.body);
    res.status(201).json(task);
  } catch (err) {
    if (err instanceof ValidationError) {
      return res.status(400).json({ error: err.message });
    }
    res.status(500).json({ error: "เกิดข้อผิดพลาดในระบบ" });
  }
});
api/src/services/task.service.jsAPI · กฎของงาน
async function createTask(input) {
  const title = (input.title || "").trim();
  if (title === "") {
    throw new ValidationError("กรุณาใส่ชื่องาน");
  }
  return taskRepository.insert({ title, status: "todo" });
}
api/src/repositories/task.repository.jsAPI → ฐานข้อมูล
async function insert(task) {
  const sql = `INSERT INTO tasks (title, status)
               VALUES ($1, $2)
               RETURNING id, title, status, created_at`;
  const result = await db.query(sql, [task.title, task.status]);
  return result.rows[0];
}

แนวคำตอบ (เปิดหลังคุยในทีมแล้วเท่านั้น)
  1. TaskForm.jsx (handleSubmit) → api/tasks.js (createTask) → tasks.routes.js (POST /tasks) → task.service.js (createTask) → task.repository.js (insert) → ตาราง tasks ใน PostgreSQL แล้วย้อนกลับทางเดิม
  2. state ชื่อ title (ตั้งค่าผ่าน setTitle ทุกครั้งที่พิมพ์)
  3. POST /api/tasks header Content-Type: application/json body {"title":"เตรียมสไลด์"}
  4. 201 Created พร้อม JSON ของงานใหม่ (มี id) หน้าจอเรียก onCreated(task) เพิ่มเข้ารายการ และล้างช่องพิมพ์
  5. หยุดที่ createTask ใน service ซึ่งโยน ValidationError → route ตอบ 400 {"error":"กรุณาใส่ชื่องาน"} → tasks.js โยน Error → TaskForm แสดง "กรุณาใส่ชื่องาน" ใต้ฟอร์ม ไม่มีอะไรถูกบันทึก
  6. route ตอบ 500 "เกิดข้อผิดพลาดในระบบ" หน้าจอแสดงข้อความนี้ใต้ฟอร์ม (ไม่บอกรายละเอียดภายในให้ผู้ใช้)
  7. แก้ที่ task.service.js เพราะเป็นที่รวมกฎของงาน ทุกทางที่สร้างงานผ่านตรงนี้ ตรวจที่หน้าจอเพิ่มได้เพื่อให้ผู้ใช้รู้เร็ว แต่ไม่พอ เพราะคำขอส่งตรงมาที่ API ได้โดยไม่ผ่านหน้าจอ
  8. architecture เช่น แยกหน้าจอ API และฐานข้อมูล คุยกันด้วย HTTP/JSON หรือใช้ PostgreSQL · design เช่น ฟังก์ชัน createTask ตรวจชื่องานว่าง หรือ SQL INSERT ที่คืน id
